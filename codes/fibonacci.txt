program fibonacci;
var n, anterior1, anterior2, soma, i: integer;

begin
	read(n);

	anterior1 := 1;
	anterior2 := 0;

	if (n <> 0 and n <> 1) then
		for i := 2 to n do
			begin
				soma := anterior1 + anterior2;
				anterior2 := anterior1;
				anterior1 := soma;
			end
	else
		if n = 0 then
			soma := anterior2
		else
			soma := anterior1;

	writeln(soma);
end.
